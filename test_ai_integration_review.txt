router = APIRouter(
    prefix="/api/healing",
    tags=["Healing Approvals"],
    dependencies=[Depends(get_current_user)],

@router.get("/requests")
def list_requests(
    status: str | None = None,
    current_user: dict = Depends(get_current_user),
):
    if status and status.upper() not in (
        "PENDING",
        "APPROVED",
        "REJECTED",
    ):
        raise HTTPException(
            status_code=400,
            detail="Invalid status filter.",
        )

    return {
        "requests": get_approval_requests(
            current_user,
            status,
        )
    }


@router.get("/requests/{request_id}")
def read_request(
    request_id: str,
    current_user: dict = Depends(get_current_user),
):
    request = get_approval_request(
        request_id,
        current_user,
    )

    if not request:
        raise HTTPException(
            status_code=404,
            detail="Approval request not found.",
        )

    return request

from datetime import datetime, timezone

from bson import ObjectId
from pymongo import ReturnDocument

from .database import (
    projects_collection,
    test_cases_collection,
    test_runs_collection,
)


def utc_now():
    return datetime.now(timezone.utc)


def approval_collection():
    return test_runs_collection.database["healing_approvals"]


def serialize_request(document):
    if not document:
        return None

    result = dict(document)
    result["id"] = str(result.pop("_id"))

    for key in ("created_at", "updated_at", "decided_at"):
        if isinstance(result.get(key), datetime):
            result[key] = result[key].isoformat()

    if result.get("test_case_id") is not None:
        result["test_case_id"] = str(result["test_case_id"])

    if result.get("project_id") is not None:
        result["project_id"] = str(result["project_id"])

    return result


def owned_tests(current_user):
    projects = list(
        projects_collection.find(
            {"owner_id": current_user["id"]},
            {"_id": 1},
        )
    )

    project_ids = [item["_id"] for item in projects]

    if not project_ids:
        return {}

    project_keys = project_ids + [
        str(item) for item in project_ids
    ]

    tests = test_cases_collection.find(
        {"project_id": {"$in": project_keys}},
        {"_id": 1, "project_id": 1, "name": 1},
    )

    return {
        str(test["_id"]): test
        for test in tests
    }


def sync_approval_requests(current_user):
    """
    Discover review-required events belonging to the user.
    Existing approval decisions are never overwritten.
    """

    tests = owned_tests(current_user)

    if not tests:
        return 0

    test_keys = []

    for test_id in tests:
        test_keys.extend([test_id, ObjectId(test_id)])

    runs = test_runs_collection.find(
        {
            "test_case_id": {"$in": test_keys},
            "approval_required_count": {"$gt": 0},
        },
        {
            "test_case_id": 1,
            "test_name": 1,
            "healing_history": 1,
        },
    )

    collection = approval_collection()
    created = 0

    for run in runs:
        test_id = str(run.get("test_case_id"))
        test = tests.get(test_id)

        if not test:
            continue

        for index, event in enumerate(
            run.get("healing_history") or []
        ):
            if not isinstance(event, dict):
                continue

            if event.get("resolution_status") != "REVIEW_REQUIRED":
                continue

            event_key = f"{run['_id']}:{index}"

            document = {
                "event_key": event_key,
                "run_id": str(run["_id"]),
                "test_case_id": test_id,
                "project_id": str(test["project_id"]),
                "test_name": run.get("test_name") or test.get("name"),
                "owner_id": current_user["id"],
                "source": event.get("source"),
                "step": event.get("step"),
                "verification": event.get("verification"),
                "original_locator": event.get("original_locator"),
                "replacement_locator": event.get("replacement_locator"),
                "confidence": event.get("confidence"),
                "reason": event.get("reason"),
                "ai_model": event.get("ai_model"),
                "ai_similarity": event.get("ai_similarity"),
                "ai_margin": event.get("ai_margin"),
                "healing_decision": event.get("healing_decision"),
                "healing_decision_reason": event.get(
                    "healing_decision_reason"
                ),
                "status": "PENDING",
                "created_at": utc_now(),
                "updated_at": utc_now(),
            }

            result = collection.update_one(
                {"event_key": event_key},
                {"$setOnInsert": document},
                upsert=True,
            )

            if result.upserted_id:
                created += 1

    return created


def get_approval_requests(current_user, status=None):
    sync_approval_requests(current_user)

    tests = owned_tests(current_user)

    if not tests:
        return []

    query = {
        "test_case_id": {"$in": list(tests.keys())}
    }

    if status:
        query["status"] = status.upper()

    documents = approval_collection().find(query).sort(
        "created_at", -1
    ).limit(500)

    return [
        serialize_request(doc)
        for doc in documents
        if str(doc.get("test_case_id")) in tests
    ]


def get_approval_request(request_id, current_user):
    if not ObjectId.is_valid(request_id):
        return None

    tests = owned_tests(current_user)

    document = approval_collection().find_one({
        "_id": ObjectId(request_id),
        "test_case_id": {"$in": list(tests.keys())},
    })

    return serialize_request(document)


def decide_approval(request_id, decision, current_user):
    if not ObjectId.is_valid(request_id):
        return None

    if decision not in ("APPROVED", "REJECTED"):
        raise ValueError("Invalid approval decision.")

    tests = owned_tests(current_user)

    document = approval_collection().find_one_and_update(
        {
            "_id": ObjectId(request_id),
            "test_case_id": {"$in": list(tests.keys())},
            "status": "PENDING",
        },
        {
            "$set": {
                "status": decision,
                "reviewed_by": current_user["name"],
                "reviewed_by_user_id": current_user["id"],
                "decided_at": utc_now(),
                "updated_at": utc_now(),
            }
        },
        return_document=ReturnDocument.AFTER,
    )

    return serialize_request(document)

def locator_to_string(target: dict | None) -> str:
    """
    Convert recorded target information into readable text.
    """

    if not isinstance(target, dict) or not target:
        return "unknown"

    parts = []

    for key in [
        "role",
        "name",
        "label",
        "placeholder",
        "id",
        "nameAttr",
        "text",
        "css",
        "tag",
    ]:
        value = target.get(key)

        if value:
            parts.append(f"{key}={value}")

    return ", ".join(parts) if parts else "unknown"

# CSS ESCAPING
# ============================================================

def escape_attribute(value: str) -> str:
    """
    Escape values used inside CSS attribute selectors.
    """

    return (
        str(value)
        .replace("\\", "\\\\")
        .replace('"', '\\"')
    )


def escape_css(value: str) -> str:
    """
    Escape a CSS identifier.
    """

    value = str(value)
    result = ""

    for char in value:
        if char.isalnum() or char in ["-", "_"]:
            result += char
        else:
            result += "\\" + format(ord(char), "x") + " "

    return result

def locator_exists(locator, timeout=PRIMARY_WAIT_MS) -> bool:
    """
    Wait for an element to become visible.

    Unlike an immediate count(), this allows time for
    asynchronous page rendering.
    """

    try:
        locator.first.wait_for(
            state="visible",
            timeout=timeout,
        )
        return True

    except Exception:
        return False


def unique_visible_locator(locator, timeout=FALLBACK_WAIT_MS):
    """
    Return a locator only if exactly one element matches
    and that element becomes visible.
    """

    try:
        locator.first.wait_for(
            state="visible",
            timeout=timeout,
        )

        if locator.count() == 1:
            return locator.first

    except Exception:
        pass

    return None

# PRIMARY LOCATOR

def build_primary_locator(page, target: dict):
    """
    Build the original locator from recorded information.

    Priority:
    1. Role + accessible name
    2. Label
    3. ID
    4. Placeholder
    5. Name attribute
    6. CSS selector
    7. Text
    8. Tag
    """

    if not isinstance(target, dict) or not target:
        return None, "No target information"

    role = target.get("role")
    name = target.get("name")

    if role and name:
        return (
            page.get_by_role(
                str(role),
                name=str(name),
                exact=True,
            ),
            f"role={role}, name={name}",
        )

    label = target.get("label")

    if label:
        return (
            page.get_by_label(
                str(label),
                exact=True,
            ),
            f"label={label}",
        )

    element_id = target.get("id")

    if element_id:
        return (
            page.locator(
                f"#{escape_css(element_id)}"
            ),
            f"id={element_id}",
        )

    placeholder = target.get("placeholder")

    if placeholder:
        return (
            page.get_by_placeholder(
                str(placeholder),
                exact=True,
            ),
            f"placeholder={placeholder}",
        )

    name_attr = target.get("nameAttr")

    if name_attr:
        return (
            page.locator(
                f'[name="{escape_attribute(name_attr)}"]'
            ),
            f"name={name_attr}",
        )

    css = target.get("css")

    if css:
        return (
            page.locator(str(css)),
            f"css={css}",
        )

    text = target.get("text") or name

    if text:
        return (
            page.get_by_text(
                str(text),
                exact=True,
            ),
            f"text={text}",
        )

    tag = target.get("tag")

    if tag:
        return (
            page.locator(str(tag)),
            f"tag={tag}",
        )

    return None, "No usable locator"

def build_fallback_candidates(page, target: dict):
    """
    Generate alternative locator strategies.

    Each candidate includes:
    (locator, description, confidence, reason)
    """

    candidates = []

    if not isinstance(target, dict):
        return candidates

    role = target.get("role")
    name = target.get("name")
    label = target.get("label")
    placeholder = target.get("placeholder")
    element_id = target.get("id")
    name_attr = target.get("nameAttr")
    css = target.get("css")
    text = target.get("text")
    tag = target.get("tag")

    # --------------------------------------------------------
    # ACCESSIBILITY ROLE
    # --------------------------------------------------------

    if role and name:
        candidates.append((
            page.get_by_role(
                str(role),
                name=str(name),
                exact=True,
            ),
            f"role={role}, name={name}",
            0.98,
            "Matched original accessibility role and name",
        ))

        candidates.append((
            page.get_by_role(
                str(role),
                name=str(name),
                exact=False,
            ),
            f"role={role}, name~={name}",
            0.92,
            "Matched accessibility role with similar name",
        ))

    # --------------------------------------------------------
    # LABEL
    # --------------------------------------------------------

    if label:
        candidates.append((
            page.get_by_label(
                str(label),
                exact=True,
            ),
            f"label={label}",
            0.96,
            "Matched form label",
        ))

        candidates.append((
            page.get_by_label(
                str(label),
                exact=False,
            ),
            f"label~={label}",
            0.88,
            "Matched similar form label",
        ))

    # --------------------------------------------------------
    # ID
    # --------------------------------------------------------

    if element_id:
        candidates.append((
            page.locator(
                f"#{escape_css(element_id)}"
            ),
            f"id={element_id}",
            0.95,
            "Matched element ID",
        ))

    # --------------------------------------------------------
    # PLACEHOLDER
    # --------------------------------------------------------

    if placeholder:
        candidates.append((
            page.get_by_placeholder(
                str(placeholder),
                exact=True,
            ),
            f"placeholder={placeholder}",
            0.94,
            "Matched exact placeholder",
        ))

        candidates.append((
            page.get_by_placeholder(
                str(placeholder),
                exact=False,
            ),
            f"placeholder~={placeholder}",
            0.86,
            "Matched similar placeholder",
        ))

    # --------------------------------------------------------
    # NAME ATTRIBUTE
    # --------------------------------------------------------

    if name_attr:
        candidates.append((
            page.locator(
                f'[name="{escape_attribute(name_attr)}"]'
            ),
            f"name={name_attr}",
            0.90,
            "Matched name attribute",
        ))

    # --------------------------------------------------------
    # VISIBLE TEXT
    # --------------------------------------------------------

    visible_text = text or name

    if visible_text:
        candidates.append((
            page.get_by_text(
                str(visible_text),
                exact=True,
            ),
            f"text={visible_text}",
            0.90,
            "Matched exact visible text",
        ))

        candidates.append((
            page.get_by_text(
                str(visible_text),
                exact=False,
            ),
            f"text~={visible_text}",
            0.82,
            "Matched similar visible text",
        ))

    # --------------------------------------------------------
    # CSS SELECTOR
    # --------------------------------------------------------

    if css:
        candidates.append((
            page.locator(str(css)),
            f"css={css}",
            0.80,
            "Matched recorded CSS selector",
        ))

    # --------------------------------------------------------
    # TAG + TEXT
    # --------------------------------------------------------

    if tag and visible_text:
        candidates.append((
            page.locator(str(tag)).filter(
                has_text=str(visible_text)
            ),
            f"{tag}:has-text({visible_text})",
            0.78,
            "Matched tag and visible text",
        ))

    return candidates

def find_element(page, target: dict) -> dict[str, Any]:
    """
    Locate a recorded element.

    First:
        Try the original recorded locator.

    Then:
        Try safe alternative locator strategies.

    Returns a result compatible with runner.py.
    """

    primary, primary_description = build_primary_locator(
        page,
        target,
    )

    # --------------------------------------------------------
    # ORIGINAL LOCATOR
    # --------------------------------------------------------

    if primary is not None:
        original_match = unique_visible_locator(
            primary,
            timeout=PRIMARY_WAIT_MS,
        )

        if original_match is not None:
            return {
                "locator": original_match,
                "healed": False,
                "original_locator": primary_description,
                "replacement_locator": None,
                "confidence": 1.0,
                "reason": "Original locator matched successfully",
            }

    # --------------------------------------------------------
    # FALLBACK SELF-HEALING
    # --------------------------------------------------------

    candidates = build_fallback_candidates(
        page,
        target,
    )

    for (
        locator,
        description,
        confidence,
        reason,
    ) in candidates:

        try:
            match = unique_visible_locator(
                locator,
                timeout=FALLBACK_WAIT_MS,
            )

            if match is not None:
                # TEST_AI_UPGRADE_1
                # Validate recorded container context.
                similarity = None
                recorded_context = str(
                    target.get("containerText") or ""
                ).strip()

                if recorded_context:
                    try:
                        live_context = match.evaluate(
                            """el => {
                                const p = el.closest(
                                    '[data-testid*="product"], article, li, [class*="product"], [class*="card"]'
                                );
                                return (
                                    p || el.parentElement || el
                                ).innerText || '';
                            }"""
                        )

                        def tokens(text):
                            import re
                            return set(
                                re.findall(
                                    r"[a-z0-9]{3,}",
                                    str(text).lower()
                                )
                            )

                        expected_tokens = tokens(
                            recorded_context
                        )
                        actual_tokens = tokens(
                            live_context
                        )

                        if expected_tokens:
                            similarity = len(
                                expected_tokens & actual_tokens
                            ) / len(expected_tokens)

                            if similarity < 0.6:
                                continue

                    except Exception:
                        continue

                # PHASE 2: Score the candidate after context validation.
                decision = evaluate_healing_candidate(
                    base_confidence=confidence,
                    context_similarity=similarity,
                    unique=True,
                )

                if decision.status == "REJECTED":
                    continue

                # PHASE 2: Approval safety gate.
                # Do not execute a candidate that requires review.
                if decision.status == "REVIEW_REQUIRED":
                    return {
                        "locator": None,
                        "healed": False,
                        "resolution_status": "REVIEW_REQUIRED",
                        "original_locator": primary_description,
                        "replacement_locator": description,
                        "confidence": decision.confidence,
                        "healing_decision": decision.status,
                        "healing_decision_reason": decision.reason,
                        "reason": (
                            "A fallback locator was found, but "
                            "human approval is required before "
                            "the test can execute this action."
                        ),
                    }

                # Avoid reporting an identical locator as healed.
                if description == primary_description:
                    return {
                        "locator": match,
                        "healed": False,
                        "original_locator": primary_description,
                        "replacement_locator": None,
                        "confidence": 1.0,
                        "reason": "Original locator matched after waiting",
                    }

                return {
                    "locator": match,
                    "healed": False,
                    "resolution_status": "FALLBACK",
                    "original_locator": primary_description,
                    "replacement_locator": description,
                    "confidence": decision.confidence,
                    "healing_decision": decision.status,
                    "healing_decision_reason": decision.reason,
                    "reason": "Existing recorded locator fallback: " + reason,
                }

        except Exception:
            continue


    # ========================================================
    # TEST_AI_SEMANTIC_HEALING_V1
    # AI-ASSISTED RECOVERY
    # ========================================================

    # Existing recorded locators have already been tried.
    # AI now examines live DOM candidates for semantic matches.
    #
    # AI suggestions are never executed automatically here.
    # They must pass further validation and human approval.

    try:
        from .ai_semantic_healing import (
            rank_semantic_candidates,
            evaluate_semantic_recovery,
        )

        original = {
            "role": target.get("role"),
            "name": target.get("name"),
            "label": target.get("label"),
            "placeholder": target.get("placeholder"),
            "text": target.get("text"),
            "tag": target.get("tag"),
            "type": target.get("type"),
        }

        selector = (
            "button, a, input:not([type='password']), "
            "select, textarea, [role='button'], "
            "[role='link'], [role='textbox']"
        )

        elements = page.locator(selector)

        live_candidates = elements.evaluate_all(
            """nodes => nodes.slice(0, 150).map((el, index) => {
                const style = window.getComputedStyle(el);
                const rect = el.getBoundingClientRect();

                const visible =
                    style.display !== 'none' &&
                    style.visibility !== 'hidden' &&
                    rect.width > 0 &&
                    rect.height > 0;

                const type = (
                    el.getAttribute('type') || ''
                ).toLowerCase();

                const sensitive =
                    type === 'password' ||
                    type === 'hidden' ||
                    type === 'file';

                const label =
                    el.labels && el.labels.length
                    ? el.labels[0].innerText
                    : '';

                return {
                    index,
                    visible: visible && !sensitive,
                    role: (
                        el.getAttribute('role') ||
                        (el.tagName.toLowerCase() === 'button'
                            ? 'button'
                            : el.tagName.toLowerCase() === 'a'
                            ? 'link'
                            : el.tagName.toLowerCase() === 'select'
                            ? 'combobox'
                            : ['input', 'textarea'].includes(
                                el.tagName.toLowerCase()
                              )
                            ? 'textbox'
                            : '')
                    ),
                    name: (
                        el.getAttribute('aria-label') ||
                        (el.innerText || '').trim()
                    ).slice(0, 120),
                    label: String(label || '').slice(0, 120),
                    placeholder: (
                        el.getAttribute('placeholder') || ''
                    ).slice(0, 120),
                    text: (
                        el.getAttribute('aria-label')
                            ? (el.innerText || '').slice(0, 120)
                            : ''
                    ),
                    tag: el.tagName.toLowerCase(),
                    type,
                };
            })"""
        )

        live_candidates = [
            item
            for item in live_candidates
            if item.get("visible")
        ]

        if live_candidates:
            ranked = rank_semantic_candidates(
                original,
                live_candidates,
                limit=5,
            )

            decision = evaluate_semantic_recovery(
                ranked,
                minimum_similarity=0.80,
                minimum_margin=0.08,
            )

            if decision["status"] == "VALIDATION_REQUIRED":
                suggested = decision["recommended_candidate"]

                suggested_index = suggested.get("index")

                if isinstance(suggested_index, int):
                    proposed_locator = elements.nth(
                        suggested_index
                    )

                    # Confirm that the proposed element
                    # still exists and is visible.
                    if (
                        proposed_locator.count() == 1
                        and proposed_locator.is_visible()
                    ):
                        return {
                            "locator": None,
                            "healed": False,
                            "resolution_status": "REVIEW_REQUIRED",
                            "original_locator": primary_description,
                            "replacement_locator": (
                                f"AI candidate: {selector}, "
                                f"index={suggested_index}"
                            ),
                            "confidence": decision[
                                "semantic_similarity"
                            ],
                            "healing_decision": "REVIEW_REQUIRED",
                            "healing_decision_reason": (
                                "AI semantic match found. "
                                "Human approval and stable locator "
                                "validation are required."
                            ),
                            "reason": (
                                "AI identified a possible replacement "
                                "element after recorded locators failed."
                            ),
                            "ai_model": decision["model"],
                            "ai_similarity": decision[
                                "semantic_similarity"
                            ],
                            "ai_margin": decision["margin"],
                        }

    except Exception as ai_error:
        # AI failure must not authorize an unsafe action.
        # Preserve normal locator failure behavior.
        import logging

        logging.getLogger(__name__).exception(
            "AI semantic recovery failed: %s",
            ai_error,
        )

    # --------------------------------------------------------
    # NO MATCH FOUND
    # --------------------------------------------------------

    return {
        "locator": None,
        "healed": False,
        "original_locator": primary_description,
        "replacement_locator": None,
        "confidence": 0.0,
        "reason": (
            "Recorded element could not be located "
            "using the original or alternative selectors."
        ),
    }

def valid_id(value: str) -> ObjectId:
    if not ObjectId.is_valid(value):
        raise HTTPException(
            status_code=400,
            detail="Invalid ID",
        )
    return ObjectId(value)


def serialize_datetime(value):
    if isinstance(value, datetime):
        return value.isoformat()
    return value


def evidence_url_from_path(path_value):
    if not path_value:
        return None

    value = str(path_value)

    if value.startswith("/evidence/"):
        return value

    filename = Path(value.replace("\\", "/")).name

    if not filename:
        return None

    # These URLs are retained for compatibility with
    # existing reports. Public evidence serving remains
    # disabled until authenticated retrieval is implemented.
    return f"/evidence/{filename}"


def normalize_status(item: dict) -> str:
    status = str(item.get("status") or "").upper()

    if status in {"PASS", "PASSED", "SUCCESS"}:
        return "PASS"

    if status == "HEALED":
        return "HEALED"

    if status in {"SKIPPED", "PENDING"}:
        return "SKIPPED"

    if status in {"FAIL", "FAILED", "ERROR"}:
        return "FAIL"

    if item.get("passed") is True:
        return "HEALED" if item.get("healed") else "PASS"

    return "FAIL"


def normalize_step(step):
    if not isinstance(step, dict):
        return {
            "status": "FAIL",
            "message": str(step),
        }

    normalized = dict(step)
    normalized["status"] = normalize_status(step)

    screenshot = (
        step.get("screenshot")
        or step.get("screenshot_path")
        or step.get("evidence_path")
    )

    if screenshot:
        normalized["screenshot"] = evidence_url_from_path(
            screenshot
        )

    for key in ("started_at", "finished_at"):
        if key in normalized:
            normalized[key] = serialize_datetime(
                normalized[key]
            )

    healing = step.get("healing")

    if isinstance(healing, dict):
        normalized["healing"] = {
            "healed": bool(healing.get("healed", True)),
            "original_locator": healing.get("original_locator"),
            "replacement_locator": healing.get("replacement_locator"),
            "confidence": healing.get("confidence"),
            "reason": healing.get("reason"),
        }

    return normalized


def normalize_verification(verification):
    if not isinstance(verification, dict):
        return {
            "status": "FAIL",
            "passed": False,
            "message": str(verification),
        }

    normalized = dict(verification)
    normalized["status"] = normalize_status(verification)

    if "passed" not in normalized:
        normalized["passed"] = normalized["status"] in {
            "PASS",
            "HEALED",
        }

    screenshot = (
        verification.get("screenshot")
        or verification.get("screenshot_path")
        or verification.get("evidence_path")
    )

    if screenshot:
        normalized["screenshot"] = evidence_url_from_path(
            screenshot
        )

    for key in ("started_at", "finished_at"):
        if key in normalized:
            normalized[key] = serialize_datetime(
                normalized[key]
            )

    return normalized


def serialize_run(run: dict) -> dict:
    mongo_id = str(run["_id"])

    test_case_id = run.get("test_case_id")

    if test_case_id is not None:
        test_case_id = str(test_case_id)

    steps = [
        normalize_step(step)
        for step in (run.get("steps") or [])
    ]

    verifications = [
        normalize_verification(item)
        for item in (run.get("verifications") or [])
    ]

    evidence = []

    for item in (run.get("evidence") or []):
        if not isinstance(item, dict):
            continue

        copy = dict(item)
        url = copy.get("url")

        if url:
            copy["url"] = evidence_url_from_path(url)
            evidence.append(copy)

    for index, step in enumerate(steps, start=1):
        screenshot = step.get("screenshot")

        if screenshot:
            evidence.append({
                "type": "step_screenshot",
                "step": index,
                "url": screenshot,
                "name": Path(screenshot).name,
            })

    for index, verification in enumerate(
        verifications,
        start=1,
    ):
        screenshot = verification.get("screenshot")

        if screenshot:
            evidence.append({
                "type": "verification_screenshot",
                "verification": index,
                "url": screenshot,
                "name": Path(screenshot).name,
            })

    unique_evidence = []
    seen = set()

    for item in evidence:
        url = item.get("url")

        if not url or url in seen:
            continue

        seen.add(url)
        unique_evidence.append(item)

    healing_details = run.get("healing_details") or []

    healing_count = run.get("healing_count")

    if healing_count is None:
        healing_count = len(healing_details)

    healed = bool(
        run.get("healed", healing_count > 0)
    )

    return {
        "id": mongo_id,
        "test_run_id": run.get("run_id") or mongo_id,
        "test_case_id": test_case_id,
        "test_name": run.get("test_name", "Unnamed Test"),
        "target_url": run.get("target_url", ""),
        "status": normalize_status(run),
        "message": run.get("message", ""),
        "started_at": serialize_datetime(
            run.get("started_at")
        ),
        "finished_at": serialize_datetime(
            run.get("finished_at")
        ),
        "duration_ms": run.get("duration_ms"),
        "steps": steps,
        "verifications": verifications,
        "evidence": unique_evidence,
        "screenshot": evidence_url_from_path(run.get("screenshot")),
        "healed": healed,
        "healing_count": healing_count,
        "approval_required_count": run.get(
            "approval_required_count", 0
        ),
        "healing_details": healing_details,
        "error": run.get("error"),
        "created_at": serialize_datetime(
            run.get("created_at")
        ),
    }

def find_test_case(test_case_id):
    if not test_case_id:
        return None

    try:
        object_id = ObjectId(str(test_case_id))
    except (ValueError, TypeError):
        return None

    return test_cases_collection.find_one({
        "_id": object_id,
    })


def owned_project_ids(current_user: dict):
    return [
        project["_id"]
        for project in projects_collection.find(
            {"owner_id": current_user["id"]},
            {"_id": 1},
        )
    ]


def owned_test_ids(current_user: dict):
    project_ids = owned_project_ids(current_user)

    if not project_ids:
        return []

    project_keys = project_ids + [
        str(project_id)
        for project_id in project_ids
    ]

    return [
        test["_id"]
        for test in test_cases_collection.find(
            {"project_id": {"$in": project_keys}},
            {"_id": 1},
        )
    ]


def run_query_for_user(current_user: dict):
    test_ids = owned_test_ids(current_user)

    if not test_ids:
        return {"_id": {"$in": []}}

    return {
        "test_case_id": {
            "$in": test_ids + [
                str(test_id)
                for test_id in test_ids
            ]
        }
    }


def require_owned_run(run_id: str, current_user: dict):
    object_id = valid_id(run_id)

    query = run_query_for_user(current_user)
    query["_id"] = object_id

    run = test_runs_collection.find_one(query)

    if run is None:
        raise HTTPException(
            status_code=404,
            detail="Test run not found",
        )

    return run


def enrich_run(run):
    response = serialize_run(run)

    test_case = find_test_case(
        run.get("test_case_id")
    )

    if test_case:
        response["test"] = {
            "id": str(test_case["_id"]),
            "name": test_case.get("name", ""),
            "target_url": test_case.get("target_url", ""),
        }

        response["test_name"] = test_case.get(
            "name",
            response["test_name"],
        )

        response["target_url"] = test_case.get(
            "target_url",
            response["target_url"],
        )

    return response

        .sort("started_at", -1)
        .limit(500)
    ]

    return {

        "total": len(runs),
    }

@router.get("/{run_id}/evidence/{filename}")
def get_run_evidence(
    run_id: str,
    filename: str,
    current_user: dict = Depends(get_current_user),
):
    run = require_owned_run(run_id, current_user)

    if (
        not filename
        or filename in {".", ".."}
        or Path(filename).name != filename
        or "\\" in filename
        or not filename.lower().endswith(".png")
    ):
        raise HTTPException(
            status_code=404,
            detail="Evidence not found",
        )

    def stored_filename(value):
        if not isinstance(value, str) or not value:
            return None
        return Path(value.replace("\\", "/")).name

    allowed = set()

    def add(value):
        name = stored_filename(value)
        if name:
            allowed.add(name)

    add(run.get("screenshot"))

    for step in run.get("steps") or []:
        if isinstance(step, dict):
            for key in (
                "screenshot",
                "screenshot_path",
                "evidence_path",
            ):
                add(step.get(key))

    for verification in run.get("verifications") or []:
        if isinstance(verification, dict):
            for key in (
                "screenshot",
                "screenshot_path",
                "evidence_path",
            ):
                add(verification.get(key))

    for item in run.get("evidence") or []:
        if isinstance(item, dict):
            for key in (
                "url",
                "screenshot",
                "path",
                "evidence_path",
            ):
                add(item.get(key))

    if filename not in allowed:
        raise HTTPException(
            status_code=404,
            detail="Evidence not found",
        )

    evidence_dir = (
        Path(__file__).resolve().parent.parent / "evidence"
    ).resolve()

    candidate = (evidence_dir / filename).resolve()

    if (
        candidate.parent != evidence_dir
        or not candidate.is_file()
    ):
        raise HTTPException(
            status_code=404,
            detail="Evidence not found",
        )

    return FileResponse(
        candidate,
        media_type="image/png",
        headers={
            "Cache-Control": "private, no-store",
            "X-Content-Type-Options": "nosniff",
        },
    )

SENSITIVE_TERMS = (
    "password",
    "passwd",
    "passcode",
    "secret",
    "token",
    "api_key",
    "authorization",
)


def is_sensitive_step(step):
    target = step.get("target")

    if not isinstance(target, dict):
        target = {}

    combined = " ".join(
        str(value).lower()
        for value in [
            target.get("name"),
            target.get("label"),
            target.get("id"),
            target.get("nameAttr"),
            target.get("placeholder"),
            target.get("type"),
            target.get("autocomplete"),
        ]
        if value
    )

    return any(
        term in combined
        for term in SENSITIVE_TERMS
    )


def safe_target_details(target):
    """
    Return diagnostic information without exposing
    recorded input values or sensitive field details.
    """
    if not isinstance(target, dict):
        return {}

    fields = [
        "role",
        "name",
        "text",
        "label",
        "placeholder",
        "id",
        "css",
        "tag",
    ]

    details = {}

    for field in fields:
        value = target.get(field)

        if value is not None and value != "":
            details[field] = str(value)[:200]

    combined = " ".join(
        str(value).lower()
        for value in target.values()
        if isinstance(value, (str, int, float))
    )

    if any(term in combined for term in SENSITIVE_TERMS):
        return {"target": "[Sensitive field]"}

    return details

class HealingApprovalRequired(AssertionError):
    def __init__(self, healing_result):
        self.healing_result = healing_result
        super().__init__(
            healing_result.get(
                "reason",
                "Locator replacement requires approval."
            )
        )

def resolve_element(page, target):
    """
    Resolve a recorded element through the self-healing
    engine.

    Provides meaningful diagnostics if the element
    cannot be located.
    """

    if not isinstance(target, dict) or not target:
        raise ValueError(
            "Recorded element target is missing or invalid."
        )

    healing_result = find_element(
        page,
        target,
    )

    if not isinstance(healing_result, dict):
        raise ValueError(
            "Invalid response from self-healing engine."
        )

    locator = healing_result.get("locator")

    if locator is None:
        if healing_result.get("resolution_status") == "REVIEW_REQUIRED":
            raise HealingApprovalRequired(healing_result)

        reason = healing_result.get(
            "reason",
            "No matching element was found.",
        )

        details = safe_target_details(target)

        current_url = page.url

        login_hint = ""

        if current_url.rstrip("/").lower().endswith(
            ("/login", "/signin", "/sign-in")
        ):
            login_hint = (
                " Browser is on a login page. "
                "Check whether an earlier step failed "
                "to establish authentication or caused "
                "an unexpected redirect."
            )

        raise AssertionError(
            "Recorded element could not be located. "
            f"Reason: {reason}. "
            f"Recorded target: {details}. "
            f"Current URL: {current_url}."
            f"{login_hint}"
        )

    return locator, healing_result


def healing_details_from(result):
    """
    Preserve Phase 2 confidence and approval metadata.

    Recorded fallback is not genuine self-healing.
    """

    if not isinstance(result, dict):
        return None

    status = result.get("resolution_status")

    if status in ("FALLBACK", "REVIEW_REQUIRED"):
        return {
            "healed": False,
            "resolution_status": status,
            "original_locator": result.get("original_locator"),
            "replacement_locator": result.get("replacement_locator"),
            "confidence": result.get("confidence"),
            "healing_decision": result.get("healing_decision"),
            "healing_decision_reason": result.get(
                "healing_decision_reason"
            ),
            "reason": result.get("reason"),
            "ai_model": result.get("ai_model"),
            "ai_similarity": result.get("ai_similarity"),
            "ai_margin": result.get("ai_margin"),
        }

    if result.get("healed"):
        return {
            "healed": True,
            "resolution_status": status or "HEALED",
            "original_locator": result.get("original_locator"),
            "replacement_locator": result.get("replacement_locator"),
            "confidence": result.get("confidence"),
            "healing_decision": result.get("healing_decision"),
            "healing_decision_reason": result.get(
                "healing_decision_reason"
            ),
            "reason": result.get("reason"),
            "ai_model": result.get("ai_model"),
            "ai_similarity": result.get("ai_similarity"),
            "ai_margin": result.get("ai_margin"),
        }

    return None

def execute_action(
    page,
    step,
    run_key,
    step_number,
):
    started_at = utc_now()
    started_clock = perf_counter()

    action = str(
        step.get("action", "")
    ).strip().lower()

    result = {
        "step": step_number,
        "action": action,
        "status": "PASS",
        "message": "",
        "started_at": started_at,
    }

    sensitive = is_sensitive_step(step)

    if not sensitive and step.get("value") is not None:
        result["value"] = str(step["value"])

    try:
        result["url_before"] = page.url

        target = step.get("target")
        value = step.get("value")

        # ----------------------------------------------------
        # NAVIGATION
        # ----------------------------------------------------

        if action in ["navigate", "goto", "open"]:
            url = str(
                step.get("url")
                or value
                or ""
            ).strip()

            if not url:
                raise ValueError(
                    "Navigation URL is missing."
                )

            if not url.startswith(("http://", "https://")):
                raise ValueError(
                    "Navigation requires an HTTP(S) URL."
                )

            page.goto(
                url,
                wait_until="domcontentloaded",
                timeout=30000,
            )

            result["url"] = page.url
            result["message"] = "Navigation completed."

        # ----------------------------------------------------
        # CLICK
        # ----------------------------------------------------

        elif action in ["click", "tap"]:
            url_before = page.url

            locator, healing = resolve_element(
                page,
                target,
            )

            locator.click(timeout=10000)

            # Wait briefly for document navigation when
            # the click has changed the current URL.
            if page.url != url_before:
                try:
                    page.wait_for_load_state(
                        "domcontentloaded",
                        timeout=3000,
                    )
                except Exception:
                    pass

            url_after = page.url

            result["url_before"] = url_before
            result["url_after"] = url_after

            if url_before != url_after:
                result["message"] = (
                    "Click completed. Navigation: "
                    f"{url_before} -> {url_after}"
                )
            else:
                result["message"] = "Click completed."

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # DOUBLE CLICK
        # ----------------------------------------------------

        elif action in ["double_click", "dblclick"]:
            locator, healing = resolve_element(
                page,
                target,
            )

            locator.dblclick(timeout=10000)

            result["message"] = "Double click completed."

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # INPUT / FILL
        # ----------------------------------------------------

        elif action in [
            "fill",
            "input",
            "type",
            "change",
        ]:
            locator, healing = resolve_element(
                page,
                target,
            )

            fill_value = str(
                value if value is not None else ""
            )

            if sensitive:
                import os

                fill_value = os.environ.get(
                    "TEST_AI_TEST_PASSWORD", ""
                )

                if not fill_value:
                    raise ValueError(
                        "Sensitive test input requires a "
                        "configured test credential."
                    )

            locator.fill(
                fill_value,
                timeout=10000,
            )

            result["message"] = (
                "Sensitive input completed."
                if sensitive
                else "Input completed."
            )

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # SELECT
        # ----------------------------------------------------

        elif action in ["select", "select_option"]:
            locator, healing = resolve_element(
                page,
                target,
            )

            if value is None:
                raise ValueError(
                    "Select action has no value."
                )

            locator.select_option(
                str(value),
                timeout=10000,
            )

            result["message"] = "Selection completed."

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # CHECK
        # ----------------------------------------------------

        elif action == "check":
            locator, healing = resolve_element(
                page,
                target,
            )

            locator.check(timeout=10000)

            result["message"] = "Checkbox checked."

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # UNCHECK
        # ----------------------------------------------------

        elif action == "uncheck":
            locator, healing = resolve_element(
                page,
                target,
            )

            locator.uncheck(timeout=10000)

            result["message"] = "Checkbox unchecked."

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # PRESS KEY
        # ----------------------------------------------------

        elif action in ["press", "keypress", "key"]:
            key = str(
                step.get("key")
                or value
                or ""
            ).strip()

            if not key:
                raise ValueError(
                    "Keyboard key is missing."
                )

            if isinstance(target, dict):
                locator, healing = resolve_element(
                    page,
                    target,
                )

                locator.press(
                    key,
                    timeout=10000,
                )

                details = healing_details_from(healing)

                if details:
                    result["status"] = "HEALED" if details.get("healed") else "PASS"
                    result["resolution_status"] = details.get("resolution_status", "HEALED")
                    result["healing"] = details

            else:
                page.keyboard.press(key)

            result["message"] = "Key pressed."

        # ----------------------------------------------------
        # WAIT
        # ----------------------------------------------------

        elif action == "wait":
            milliseconds = int(
                step.get("ms")
                or value
                or 1000
            )

            if milliseconds < 0 or milliseconds > 30000:
                raise ValueError(
                    "Wait must be between 0 and 30000 ms."
                )

            page.wait_for_timeout(milliseconds)

            result["message"] = "Wait completed."

        # ----------------------------------------------------
        # SCROLL
        # ----------------------------------------------------

        elif action == "scroll":
            x = float(step.get("x", 0))

            y = float(
                step.get("y")
                or step.get("deltaY")
                or 500
            )

            page.mouse.wheel(x, y)

            result["message"] = "Scroll completed."

        # ----------------------------------------------------
        # VERIFY ELEMENT
        # ----------------------------------------------------

        elif action in [
            "verify_element",
            "assert_element",
        ]:
            locator, healing = resolve_element(
                page,
                target,
            )

            locator.wait_for(
                state="visible",
                timeout=10000,
            )

            result["message"] = (
                "Element verification passed."
            )

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # UNKNOWN ACTION
        # ----------------------------------------------------

        else:
            raise ValueError(
                f"Unsupported test action: "
                f"{action or 'empty'}"
            )

    except Exception as error:
        result["status"] = "FAIL"
        result["error"] = str(error)

        if isinstance(error, HealingApprovalRequired):
            result["resolution_status"] = "REVIEW_REQUIRED"
            result["healing"] = healing_details_from(
                error.healing_result
            )
        result["message"] = (
            f"Step {step_number} failed."
        )

        # Save safe target information for debugging.
        if isinstance(step.get("target"), dict):
            result["target_details"] = safe_target_details(
                step["target"]
            )

        try:
            result["current_url"] = page.url
        except Exception:
            pass

        evidence = capture_screenshot(
            page,
            run_key,
            f"step_{step_number}_failed",
        )

        if evidence:
            result["screenshot"] = evidence

    finally:
        try:
            result["url_after"] = page.url
        except Exception:
            pass

        result["finished_at"] = utc_now()

        result["duration_ms"] = round(
            (perf_counter() - started_clock) * 1000,
            2,
        )

    return result

def execute_verification(
    page,
    verification,
    run_key,
    verification_number,
):
    verification_type = str(
        verification.get("type", "")
    ).strip().lower()

    condition = str(
        verification.get("condition", "contains")
    ).strip().lower()

    expected = verification.get("value")

    result = {
        "type": verification_type,
        "condition": condition,
        "status": "PASS",
    }

    if expected is not None:
        result["expected"] = str(expected)

    try:
        # ----------------------------------------------------
        # URL COMPARISON
        # ----------------------------------------------------

        if verification_type == "url":
            comparison = compare_url(
                expected=expected,
                actual_url=page.url,
                condition=condition,
            )

            result.update(comparison)

            if not comparison["matched"]:
                raise AssertionError(
                    comparison["message"]
                )

        # ----------------------------------------------------
        # PAGE TITLE COMPARISON
        # ----------------------------------------------------

        elif verification_type == "page_title":
            comparison = compare_page_title(
                expected=expected,
                actual_title=page.title(),
                condition=condition,
            )

            result.update(comparison)

            if not comparison["matched"]:
                raise AssertionError(
                    comparison["message"]
                )

        # ----------------------------------------------------
        # PAGE TEXT COMPARISON
        # ----------------------------------------------------

        elif verification_type == "text":
            actual_text = page.locator(
                "body"
            ).inner_text(timeout=10000)

            comparison = compare_page_text(
                expected=expected,
                actual_page_text=actual_text,
                condition=condition,
            )

            result.update(comparison)

            if not comparison["matched"]:
                raise AssertionError(
                    comparison["message"]
                )

        # ----------------------------------------------------
        # ELEMENT EXISTS
        # ----------------------------------------------------

        elif verification_type == "element":
            target = verification.get("target")

            if not isinstance(target, dict):
                raise ValueError(
                    "Element verification has no "
                    "recorded target."
                )

            locator, healing = resolve_element(
                page,
                target,
            )

            locator.wait_for(
                state="visible",
                timeout=10000,
            )

            result["expected"] = "Element visible"
            result["actual"] = "Element visible"
            result["matched"] = True
            result["message"] = (
                "Element verification passed."
            )

            details = healing_details_from(healing)

            if details:
                result["status"] = "HEALED" if details.get("healed") else "PASS"
                result["resolution_status"] = details.get("resolution_status", "HEALED")
                result["healing"] = details

        # ----------------------------------------------------
        # UNKNOWN VERIFICATION
        # ----------------------------------------------------

        else:
            raise ValueError(
                "Unsupported verification type: "
                f"{verification_type}"
            )

    except Exception as error:
        result["status"] = "FAIL"
        result["matched"] = False
        result["error"] = str(error)

        if isinstance(error, HealingApprovalRequired):
            result["resolution_status"] = "REVIEW_REQUIRED"
            result["healing"] = healing_details_from(
                error.healing_result
            )

        if not result.get("message"):
            result["message"] = (
                "Expected vs actual comparison failed."
            )

        evidence = capture_screenshot(
            page,
            run_key,
            f"verification_{verification_number}_failed",
        )

        if evidence:
            result["screenshot"] = evidence

    return result

def calculate_run_status(
    step_results,
    verification_results,
):
    all_results = (
        list(step_results)
        + list(verification_results)
    )

    if not all_results:
        return "FAIL"

    if any(
        result.get("status") == "FAIL"
        for result in all_results
    ):
        return "FAIL"

    if any(
        result.get("status") == "HEALED"
        for result in all_results
    ):
        return "HEALED"

    return "PASS"

def run_test(test_case):

                    # OPEN TARGET

                    # EXECUTE EXPECTED COMPARISONS

                            start=1,

function formatDate(value?: string | null): string {
  if (!value) return "Not available";

  const date = new Date(value);

  return Number.isNaN(date.getTime())
    ? "Not available"
    : date.toLocaleString();
}

function formatConfidence(value?: number | null): string {
  if (typeof value !== "number" || !Number.isFinite(value)) {
    return "Not available";
  }

  return `${(value * 100).toFixed(1)}%`;
}

function statusStyles(status: ApprovalStatus): string {
  if (status === "APPROVED") {

    return "border-rose-500/30 bg-rose-500/10 text-rose-300";
  }

  return "border-amber-500/30 bg-amber-500/10 text-amber-300";
}

"use client";

import { useCallback, useEffect, useMemo, useState } from "react";
import Link from "next/link";

type RunStatus = "PASS" | "FAIL" | "HEALED" | "SKIPPED";

interface TestRun {
  id: string;
  test_case_id?: string;
  test_name?: string;
  target_url?: string;
  status?: string;
  started_at?: string;
  finished_at?: string;
  duration_ms?: number;
  healing_count?: number;
  approval_required_count?: number;
  verifications?: Array<{
    status?: string;
    passed?: boolean;
  }>;
}

interface RunsResponse {
  runs: TestRun[];
}

function normalizeStatus(value?: string): RunStatus {
  const status = (value || "").toUpperCase();

  if (status === "PASS") return "PASS";
  if (status === "HEALED") return "HEALED";
  if (status === "SKIPPED") return "SKIPPED";

  return "FAIL";
}

function formatDate(value?: string) {
  if (!value) return "—";

  const date = new Date(value);

  return Number.isNaN(date.getTime())
    ? "—"
    : date.toLocaleString();
}

function formatDuration(value?: number) {
  if (value === undefined || value === null) {
    return "—";
  }

  if (value < 1000) {
    return `${Math.round(value)} ms`;
  }

  return `${(value / 1000).toFixed(2)} s`;
}

function StatusBadge({ status }: { status: RunStatus }) {
  const styles: Record<RunStatus, string> = {
    PASS: "border-emerald-500/30 bg-emerald-500/10 text-emerald-300",
    HEALED: "border-violet-500/30 bg-violet-500/10 text-violet-300",
    FAIL: "border-red-500/30 bg-red-500/10 text-red-300",
    SKIPPED: "border-slate-500/30 bg-slate-500/10 text-slate-300",
  };

  return (
    <span
      className={`inline-flex rounded-full border px-3 py-1 text-xs font-semibold ${styles[status]}`}
    >
      {status}
    </span>
  );
}

function MetricCard({
  title,
  value,
  subtitle,
  color = "text-white",
}: {
  title: string;
  value: string | number;
  subtitle: string;
  color?: string;
}) {
  return (
    <div className="rounded-2xl border border-white/10 bg-[#111723] p-5">
      <p className="text-sm text-slate-400">{title}</p>

      <h2 className={`mt-3 text-3xl font-bold ${color}`}>
        {value}
      </h2>

      <p className="mt-2 text-xs text-slate-500">
        {subtitle}
      </p>
    </div>
  );
}

    (sum, run) =>

    0
  );
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [filter, setFilter] = useState("ALL");

  const loadReports = useCallback(async () => {
    try {
      setLoading(true);
      setError("");

      const response = await api<RunsResponse>("/api/runs");

      setRuns(
        Array.isArray(response.runs)
          ? response.runs
          : []
      );
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to load execution reports."
      );
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void loadReports();
  }, [loadReports]);

  const stats = useMemo(() => {
    const total = runs.length;

    const passed = runs.filter(
      (run) => normalizeStatus(run.status) === "PASS"
    ).length;

    const healed = runs.filter(
      (run) => normalizeStatus(run.status) === "HEALED"
    ).length;

    const failed = runs.filter(
      (run) => normalizeStatus(run.status) === "FAIL"
    ).length;

    const skipped = runs.filter(
      (run) => normalizeStatus(run.status) === "SKIPPED"
    ).length;

    const completed = passed + healed + failed;

    const passRate = completed
      ? (((passed + healed) / completed) * 100).toFixed(1)
      : "0.0";

    const healingEvents = runs.reduce(
      (sum, run) => sum + (Number(run.healing_count) || 0),
      0
    );



    const durations = runs
      .map((run) => run.duration_ms)
      .filter(
        (value): value is number =>
          typeof value === "number" &&
          Number.isFinite(value) &&
          value >= 0
      );

    const averageDuration = durations.length
      ? durations.reduce((sum, value) => sum + value, 0) /
        durations.length
      : undefined;

    return {
      total,
      passed,
      healed,
      failed,
      skipped,
      completed,
      passRate,
      healingEvents,
      averageDuration,
    };
